{{- /* A ```mermaid block becomes the SVG that hack/diagrams/render.mjs made of it, inline, so it takes the colours of the current theme. The file is named by the sha256 of the block: a changed diagram that was not rendered again stops the build. Text equivalent: accTitle and accDescr in the diagram, an optional caption="…" on the fence. */ -}} {{- $id := substr (sha256 .Inner) 0 16 -}} {{- with resources.Get (printf "diagrams/%s.svg" $id) -}} {{- else -}} {{- errorf "%s: diagram %s has no SVG yet; run: cd hack/diagrams && npm ci && node render.mjs" $.Page.File.Path $id -}} {{- end -}}
